cmake_minimum_required(VERSION 3.20)
project(AirPlayReceiverCore LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(RECEIVER_BUILD_UPSTREAM "Build the pinned UxPlay protocol library" OFF)
if(RECEIVER_BUILD_UPSTREAM)
  find_package(OpenSSL REQUIRED)
  include(cmake/Upstream.cmake)
endif()
add_library(receiver_core Core/PTPProbe.cpp)
target_include_directories(receiver_core PUBLIC Core)
target_compile_options(receiver_core PRIVATE -Wall -Wextra -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests Tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE receiver_core)
  target_compile_options(core_tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME core_tests COMMAND core_tests)
  if(RECEIVER_BUILD_UPSTREAM AND NOT APPLE)
    add_executable(upstream_tests Tests/upstream_tests.cpp)
    target_link_libraries(upstream_tests PRIVATE uxplay)
    add_test(NAME upstream_tests COMMAND upstream_tests)
    set_tests_properties(upstream_tests PROPERTIES TIMEOUT 20)
  endif()
endif()
